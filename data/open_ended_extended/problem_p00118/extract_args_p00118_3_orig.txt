def extract_arguments(fh):
    datasets = []
    while True:
        H, W = map(int, fh.readline().strip().split())
        if H == 0 and W == 0:
            break
        orchard = [fh.readline().strip() for _ in range(H)]
        datasets.append((H, W, orchard))
    return datasets
