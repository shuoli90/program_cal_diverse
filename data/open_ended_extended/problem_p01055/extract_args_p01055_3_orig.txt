def extract_arguments(fh):
    N, H, W = map(int, fh.readline().strip().split())
    fuse_paths = []
    for _ in range(N):
        Li = int(fh.readline().strip())
        path = []
        for _ in range(Li):
            px, py = map(int, fh.readline().strip().split())
            path.append((px, py))
        fuse_paths.append(path)
    return N, H, W, fuse_paths
