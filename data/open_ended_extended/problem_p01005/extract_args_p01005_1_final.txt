def extract_arguments(fh):
    data_sets = []
    while True:
        q, l = map(int, fh.readline().strip().split())
        if q == 0 and l == 0:
            break
        queries = []
        for _ in range(q):
            queries.append(list(map(int, fh.readline().strip().split())))
        data_sets.append((q, l, queries))
    return data_sets
