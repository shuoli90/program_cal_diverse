def extract_arguments(fh):
    # Read the first line to get N and F
    N, F = map(int, fh.readline().strip().split())
    # Read the next N lines to get the NxN matrix X
    X = []
    for _ in range(N):
        X.append(list(map(int, fh.readline().strip().split())))
    return N, F, X
