def extract_arguments(fh):
    first_line = fh.readline().strip().split()
    N = int(first_line[0])
    Q = int(first_line[1])
    requests = []
    for _ in range(Q):
        request = tuple(map(int, fh.readline().strip().split()))
        requests.append(request)
    return N, Q, requests
