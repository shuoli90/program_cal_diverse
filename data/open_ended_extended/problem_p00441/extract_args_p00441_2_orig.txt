def extract_arguments(fh):
    datasets = []
    while True:
        n = int(fh.readline().strip())
        if n == 0:
            break
        pillars = []
        for _ in range(n):
            x, y = map(int, fh.readline().strip().split())
            pillars.append((x, y))
        datasets.append((n, pillars))
    return datasets
