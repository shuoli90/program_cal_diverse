def extract_arguments(fh):
    inputs = []
    while True:
        n = int(fh.readline().strip())
        if n == 0:
            break
        antennas = []
        for _ in range(n):
            x, y, r = map(float, fh.readline().strip().split())
            antennas.append((x, y, r))
        inputs.append(antennas)
    return inputs
