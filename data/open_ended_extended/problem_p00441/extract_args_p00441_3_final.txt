def extract_arguments(fh):
    datasets = []
    while True:
        n = int(fh.readline().strip())
        if n == 0:
            break
        coordinates = []
        for _ in range(n):
            x, y = map(int, fh.readline().strip().split())
            coordinates.append((x, y))
        datasets.append((n, coordinates))
    return datasets
