def extract_arguments(fh):
    first_line = fh.readline().strip().split()
    N = int(first_line[0])
    K = int(first_line[1])
    edges = []
    for _ in range(N-1):
        edge = tuple(map(int, fh.readline().strip().split()))
        edges.append(edge)
    return N, K, edges
