def tcgen():

# Function to format the test case for easy use
def format_test_case(N, L):
    return f"{N}\n{' '.join(map(str, L))}"

# Example usage
if __name__ == "__main__":
    N, L = tcgen()
    test_case = format_test_case(N, L)
    print(test_case)
