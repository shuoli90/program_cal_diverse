def extract_arguments(fh):
    H, W, N = map(int, fh.readline().strip().split())
    obstacles = []
    for _ in range(N):
        X, Y = map(int, fh.readline().strip().split())
        obstacles.append((X, Y))
    return H, W, N, obstacles
