def extract_arguments(fh):
    T = int(fh.readline().strip())
    datasets = []
    for _ in range(T):
        H, W = map(int, fh.readline().strip().split())
        map_data = [fh.readline().strip() for _ in range(H)]
        actions = fh.readline().strip()
        datasets.append((H, W, map_data, actions))
    return datasets
