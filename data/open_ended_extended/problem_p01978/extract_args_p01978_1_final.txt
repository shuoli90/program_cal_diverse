def extract_arguments(fh):
    N, M, s, t = map(int, fh.readline().strip().split())
    edges = []
    for _ in range(M):
        a, b, c, d = map(int, fh.readline().strip().split())
        edges.append((a, b, c, d))
    return N, M, s, t, edges
