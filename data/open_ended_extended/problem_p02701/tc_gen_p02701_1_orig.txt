import string

def random_string(length):

def tcgen():
    # Minimum and Maximum constraints
    N = random.randint(1, 2 * 10**5)
    
    # Generating a list of random strings
    S = [random_string(random.randint(1, 10)) for _ in range(N)]
    
    return N, S

def print_test_case():
    N, S = tcgen()
    print(N)
    for s in S:
        print(s)

# Example usage
if __name__ == "__main__":
    print_test_case()
