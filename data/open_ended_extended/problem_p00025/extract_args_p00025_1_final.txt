def extract_arguments(fh):
    inputs = []
    while True:
        line_a = fh.readline().strip()
        if not line_a:
            break
        a = list(map(int, line_a.split()))
        b = list(map(int, fh.readline().strip().split()))
        inputs.append((a, b))
    return inputs
