def extract_arguments(fh):
    inputs = []
    while True:
        n = int(fh.readline().strip())
        if n == 0:
            break
        dataset = []
        for _ in range(n):
            x, y, r = map(int, fh.readline().strip().split())
            dataset.append((x, y, r))
        inputs.append(dataset)
    return inputs
