def extract_arguments(fh):
    datasets = []
    while True:
        n = int(fh.readline().strip())
        if n == 0:
            break
        rectangles = []
        for _ in range(n):
            l, t, r, b = map(int, fh.readline().strip().split())
            rectangles.append((l, t, r, b))
        datasets.append((n, rectangles))
    return datasets
