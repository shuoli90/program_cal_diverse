def extract_arguments(fh):
    n = int(fh.readline().strip())
    nodes = []
    for _ in range(n):
        nodes.append(list(map(int, fh.readline().strip().split())))
    q = int(fh.readline().strip())
    queries = []
    for _ in range(q):
        queries.append(list(map(int, fh.readline().strip().split())))
    return n, nodes, q, queries
