def extract_arguments(fh):
    N = int(fh.readline().strip())
    A = list(map(int, fh.readline().strip().split()))
    edges = []
    for _ in range(N - 1):
        u, v = map(int, fh.readline().strip().split())
        edges.append((u, v))
    return N, A, edges
