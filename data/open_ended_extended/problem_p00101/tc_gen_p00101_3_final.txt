import random

def generate_random_text(length):

    words = text.split()
    insert_positions = random.sample(range(len(words) + 1), k=random.randint(1, min(5, len(words) + 1)))

        words.insert(pos, 'Hoshino')
    return ' '.join(words)

        # Ensure the text does not exceed 1000 characters
