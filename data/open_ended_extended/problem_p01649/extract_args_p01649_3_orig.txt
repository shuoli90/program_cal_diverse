def extract_arguments(fh):
    datasets = []
    while True:
        n = int(fh.readline().strip())
        if n == 0:
            break
        w, h, r, v_x, v_y = map(int, fh.readline().strip().split())
        balls = [tuple(map(int, fh.readline().strip().split())) for _ in range(n)]
        datasets.append((n, w, h, r, v_x, v_y, balls))
    return datasets
