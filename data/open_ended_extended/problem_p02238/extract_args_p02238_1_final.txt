def extract_arguments(fh):
    n = int(fh.readline().strip())
    adjacency_list = []
    for _ in range(n):
        parts = list(map(int, fh.readline().strip().split()))
        u = parts[0]
        k = parts[1]
        v = parts[2:]
        adjacency_list.append((u, k, v))
    return n, adjacency_list
