def extract_arguments(fh):
    inputs = []
    while True:
        line = fh.readline().strip()
        if line:
            y1, m1, d1, y2, m2, d2 = map(int, line.split())
            if y1 < 0 or m1 < 0 or d1 < 0 or y2 < 0 or m2 < 0 or d2 < 0:
                break
            inputs.append((y1, m1, d1, y2, m2, d2))
        else:
            break
    return inputs
