def extract_arguments(fh):
    data_sets = []
    sequence_number = 1
    while True:
        n = int(fh.readline().strip())
        if n == 0:
            break
        antennas = []
        for _ in range(n):
            x, y, r = map(float, fh.readline().strip().split())
            antennas.append((x, y, r))
        data_sets.append((sequence_number, antennas))
        sequence_number += 1
    return data_sets
