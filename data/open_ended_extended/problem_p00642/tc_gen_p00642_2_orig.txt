import random

def tcgen():
    test_cases = []
    
    # Number of datasets to generate
    num_datasets = random.randint(1, 100)
    
    for _ in range(num_datasets):
        # Generate a random n value within the given constraints
        n = random.randint(1, 100000)
        test_cases.append(n)
    
    # Add the terminating condition
    test_cases.append(0)
    
    return test_cases
