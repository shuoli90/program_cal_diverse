def extract_arguments(fh):
    first_line = list(map(int, fh.readline().strip().split()))
    N = first_line[0]
    F = first_line[1]
    X = []
    for _ in range(N):
        row = list(map(int, fh.readline().strip().split()))
        X.append(row)
    return N, F, X
