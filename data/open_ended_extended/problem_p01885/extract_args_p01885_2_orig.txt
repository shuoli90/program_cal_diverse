def extract_arguments(fh):
    N, L = map(int, fh.readline().strip().split())
    A = []
    B = []
    for _ in range(N):
        a, b = map(int, fh.readline().strip().split())
        A.append(a)
        B.append(b)
    C = []
    for _ in range(N):
        c = int(fh.readline().strip())
        C.append(c)
    return N, L, A, B, C
