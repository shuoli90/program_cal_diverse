def extract_arguments(fh):
    N = int(fh.readline().strip())
    H, W = map(int, fh.readline().strip().split())
    paths = []
    for _ in range(N):
        Li = int(fh.readline().strip())
        path = []
        for _ in range(Li):
            pxij, pyij = map(int, fh.readline().strip().split())
            path.append((pxij, pyij))
        paths.append(path)
    return N, H, W, paths
