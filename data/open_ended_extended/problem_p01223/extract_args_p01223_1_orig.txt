def extract_arguments(fh):
    t = int(fh.readline().strip())
    datasets = []
    for _ in range(t):
        n = int(fh.readline().strip())
        heights = list(map(int, fh.readline().strip().split()))
        datasets.append((n, heights))
    return datasets
