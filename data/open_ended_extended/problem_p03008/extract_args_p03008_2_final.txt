def extract_arguments(fh):
    N = int(fh.readline().strip())
    g_A, s_A, b_A = map(int, fh.readline().strip().split())
    g_B, s_B, b_B = map(int, fh.readline().strip().split())
    return N, (g_A, s_A, b_A), (g_B, s_B, b_B)
