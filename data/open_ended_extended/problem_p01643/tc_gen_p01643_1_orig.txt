import random

def generate_permutation(n):
    """
    Generate a valid permutation where a_i != i and a_{a_i} = i.
    """
    p = list(range(1, n+1))
    while True:
        random.shuffle(p)
        valid = True
        for i in range(n):
            if p[i] == i+1 or p[p[i]-1] != i+1:
                valid = False
                break
        if valid:
            return p
