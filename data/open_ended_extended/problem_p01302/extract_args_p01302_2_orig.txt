def extract_arguments(fh):
    datasets = []
    while True:
        n = int(fh.readline().strip())
        if n == 0:
            break
        vx, vy = map(int, fh.readline().strip().split())
        vertices = [tuple(map(int, fh.readline().strip().split())) for _ in range(n)]
        datasets.append((n, (vx, vy), vertices))
    return datasets
