def extract_arguments(fh):
    N = int(fh.readline().strip())
    areas = []
    for _ in range(N):
        points = int(fh.readline().strip())
        polygon = [tuple(map(int, fh.readline().strip().split())) for _ in range(points)]
        areas.append(polygon)
    return N, areas
