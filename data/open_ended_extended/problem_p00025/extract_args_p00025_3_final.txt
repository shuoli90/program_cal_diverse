def extract_arguments(fh):
    inputs = []
    while True:
        line = fh.readline().strip()
        if not line:
            break
        A = list(map(int, line.split()))
        B = list(map(int, fh.readline().strip().split()))
        inputs.append((A, B))
    return inputs
