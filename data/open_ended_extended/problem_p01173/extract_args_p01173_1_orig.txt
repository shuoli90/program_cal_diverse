def extract_arguments(fh):
    datasets = []
    while True:
        n = int(fh.readline().strip())
        if n == 0:
            break
        vw, vc = map(int, fh.readline().strip().split())
        points = [tuple(map(int, fh.readline().strip().split())) for _ in range(n)]
        datasets.append((n, vw, vc, points))
    return datasets
